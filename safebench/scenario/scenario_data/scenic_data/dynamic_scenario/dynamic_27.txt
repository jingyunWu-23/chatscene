'''The ego commences an unprotected left turn at an intersection while yielding to an oncoming car when the adversarial car, comes from the right, blocks multiple lanes by driving extremely slowly, forcing the ego vehicle to change lanes.'''
Town = 'Town05'
param map = localPath(f'../maps/{Town}.xodr') 
param carla_map = Town
model scenic.simulators.carla.model
EGO_MODEL = "vehicle.lincoln.mkz_2017"

behavior AdvBehavior():
    while True:
        take SetSpeedAction(globalParameters.OPT_SLOW_SPEED)  # Maintains an extremely slow speed to block traffic.
        # Optionally perform minimal lateral variation to span multiple lanes (e.g., gentle swaying)
        if globalParameters.OPT_LATERAL_VARIATION:
            take SetSteerAction(globalParameters.OPT_ADV_STEER)
        for _ in range(globalParameters.OPT_WAIT_STEER):
            wait

param OPT_SLOW_SPEED = Range(0, 2)  # Extremely slow speed to impede traffic flow.
param OPT_ADV_STEER = Range(-0.3, 0.3)  # Small steering variation to drift across lane boundaries.
param OPT_WAIT_STEER = Range(10, 30)  # Duration between steering adjustments.
param OPT_LATERAL_VARIATION = Boolean()  # Enables/disables lateral swaying to block multiple lanes.

## MONITORS
monitor TrafficLights:
    freezeTrafficLights()
    while True:
        if withinDistanceToTrafficLight(ego, 100):
            setClosestTrafficLightStatus(ego, "green")
        if withinDistanceToTrafficLight(AdvAgent, 100):
            setClosestTrafficLightStatus(AdvAgent, "green")
        wait

intersection = Uniform(*filter(lambda i: i.is4Way and i.isSignalized, network.intersections))
egoInitLane = Uniform(*intersection.incomingLanes)
egoManeuver = Uniform(*filter(lambda m: m.type is ManeuverType.LEFT_TURN, egoInitLane.maneuvers))
egoTrajectory = [egoInitLane, egoManeuver.connectingLane, egoManeuver.endLane]
egoSpawnPt = OrientedPoint in egoManeuver.startLane.centerline

# Setting up the ego vehicle at the initial position
ego = Car at egoSpawnPt,
    with regionContainedIn None,
    with blueprint EGO_MODEL

require 10 <= (distance to intersection) <= 40
# Adversarial agent spawns from the right cross-traffic lane, entering the intersection
param OPT_CROSS_DISTANCE = Range(5, 25)
param OPT_CROSS_SPEED = Range(3, 12)

# Get the ego's current lane section and its right-side cross-traffic lane (perpendicular, typically via intersection)
# In CARLA/Scenic 2.1, cross-traffic lanes adjacent to ego's lane at an intersection are accessed via _crossingLanes
# Since ego is approaching an intersection, we assume a valid crossing lane exists to the right (i.e., ego's right-hand side in world frame, orthogonal to ego's heading)
crossLane = network.laneSectionAt(ego)._crossingLanes[0]  # First available crossing lane — assumed to be right-side cross traffic
# If multiple crossing lanes exist and we need specifically the *right-side* one relative to ego, fallback to geometry:
# But per Scenic 2.1 CARLA semantics, _crossingLanes are ordered by angle; we rely on network topology.
# To ensure robustness, project a point to the right of ego’s position onto the crossing lane centerline.

# Compute a spawn point along the cross lane, positioned so it enters intersection toward ego
# Use egoSpawnPt as reference, offset perpendicularly to ego's heading to approximate "right" direction in world frame
rightDir = egoSpawnPt.heading + 90 deg
crossSpawnOffset = globalParameters.OPT_CROSS_DISTANCE @ 0  # distance along cross lane direction — but we need to align with crossLane orientation
# Instead: sample a point near ego’s right side, then project onto crossLane centerline
rightOfEgo = egoSpawnPt.position offset along egoSpawnPt.heading + 90 deg by 5
projected = crossLane.centerline.project(rightOfEgo)
crossSpawnPt = Vector(*projected.coords[0])

# Ensure spawn point is upstream (i.e., before intersection) along crossLane — move backward along crossLane direction
crossHeading = crossLane.orientation[crossSpawnPt]
# Move upstream (opposite to crossLane flow direction) by OPT_CROSS_DISTANCE
upstreamOffset = -globalParameters.OPT_CROSS_DISTANCE * Vector(cos(crossHeading), sin(crossHeading))
IntSpawnPt = crossSpawnPt + upstreamOffset

# Spawn adversarial car aligned with cross lane, occupying multiple lanes (handled via width/scenario logic — not explicit in spawn)
AdvAgent = Car at IntSpawnPt,
    with heading crossHeading,
    with regionContainedIn None,
    with behavior AdvBehavior(target_speed=globalParameters.OPT_CROSS_SPEED)
