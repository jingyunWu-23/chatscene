'''The ego vehicle is attempting to change lanes to avoid a slow-moving leading vehicle; the adversarial car in the target lane suddenly slows down, matching the speed of the leading vehicle, and effectively blocking the ego vehicle from completing the lane change.'''
Town = 'Town03'
param map = localPath(f'../maps/{Town}.xodr') 
param carla_map = Town
model scenic.simulators.carla.model
EGO_MODEL = "vehicle.lincoln.mkz_2017"

param OPT_ADV_SPEED = Range(5, 10)
param OPT_ADV_DISTANCE = Range(0, 20)

behavior AdvBehavior():
    do FollowLaneBehavior(target_speed=globalParameters.OPT_ADV_SPEED) until (distance to self) < globalParameters.OPT_ADV_DISTANCE
    while True:
        take SetVelocityAction(*LeadingAgent.velocity)
# Identifying lane sections with at least one adjacent lane (left or right) moving in the same forward direction
laneSecsWithAdjacentSameDir = []
for lane in network.lanes:
    for laneSec in lane.sections:
        hasAdjacentSameDir = (
            (laneSec._laneToLeft is not None and laneSec._laneToLeft.isForward == laneSec.isForward) or
            (laneSec._laneToRight is not None and laneSec._laneToRight.isForward == laneSec.isForward)
        )
        if hasAdjacentSameDir:
            laneSecsWithAdjacentSameDir.append(laneSec)

# Selecting a random lane section from identified sections for the ego vehicle
egoLaneSec = Uniform(*laneSecsWithAdjacentSameDir)
egoSpawnPt = OrientedPoint in egoLaneSec.centerline

# Ego vehicle setup
ego = Car at egoSpawnPt,
    with regionContainedIn None,
    with blueprint EGO_MODEL
# Identifying the target lane (intended merge lane) relative to ego's current lane
param OPT_MERGE_DISTANCE = Range(0, 30)
advLane = network.laneSectionAt(ego)._laneToRight.lane  # assuming merge right; adjust if left is intended, but description doesn't specify direction — default to right as most common and consistent with first two examples
IntSpawnPt = OrientedPoint following roadDirection from egoSpawnPt for globalParameters.OPT_MERGE_DISTANCE
projectPt = Vector(*advLane.centerline.project(IntSpawnPt.position).coords[0])
advHeading = advLane.orientation[projectPt]

# Spawn the Adversarial Agent centered in the target lane, directly ahead of ego's intended merge position
AdvAgent = Car at projectPt,
    with heading advHeading,
    with regionContainedIn None,
    with behavior AdvBehavior()
